    (function registerPwaServiceWorker() {
      if (!('serviceWorker' in navigator)) return;
      var isHttpOrigin = window.location.protocol === 'https:' || window.location.hostname === 'localhost';
      if (isHttpOrigin) {
        navigator.serviceWorker.register('/sw.js', { scope: '/' }).catch(function () {
          registerBlobFallback();
        });
      } else {
        registerBlobFallback();
      }

      function registerBlobFallback() {
        try {
          var swLines = [
            "const CACHE_NAME = 'lms-anticurang-man4hst-v2';",
            "self.addEventListener('install', function () { self.skipWaiting(); });",
            "self.addEventListener('activate', function (e) { e.waitUntil(self.clients.claim()); });",
            "self.addEventListener('fetch', function (e) {",
            "  if (e.request.method !== 'GET') return;",
            "  e.respondWith(fetch(e.request).catch(function () { return caches.match(e.request); }));",
            "});"
          ];
          var blob = new Blob([swLines.join(String.fromCharCode(10))], { type: 'application/javascript' });
          navigator.serviceWorker.register(URL.createObjectURL(blob)).catch(function () {});
        } catch (err) {}
      }
    })();

      async function fetchAllDataFromSupabase() {
        if (IS_PREVIEW || !supabase) return;
        try {
          var results = await Promise.all([
            supabase.from('peserta').select('*').order('created_at', { ascending: false }),
            supabase.from('sesi_ujian').select('*').order('created_at', { ascending: false }),
            supabase.from('materi_info').select('*').order('created_at', { ascending: false }),
            supabase.from('monitoring_ujian').select('*').order('waktu_mulai', { ascending: false }),
            supabase.from('log_pelanggaran').select('*').order('waktu_kejadian', { ascending: false })
          ]);

          if (results[1].data) state.sesiList = results[1].data;

          if (results[4].data) state.logPelanggaranList = results[4].data;
        } catch (err) {
          console.error('Gagal memuat data Supabase:', err);

        }
      }

      async function initAuthSession() {
        var previewBanner = document.getElementById('preview-banner');
        if (IS_PREVIEW && previewBanner) {
          previewBanner.classList.remove('hidden');
          previewBanner.classList.add('flex');
        } else if (previewBanner) {
          previewBanner.remove();
        }

        if (!IS_PREVIEW && supabase) {
          try {
            var sessionRes = await supabase.auth.getSession();
            if (sessionRes.data && sessionRes.data.session && sessionRes.data.session.user) {
              var authUser = sessionRes.data.session.user;
              state.user = {
                role: 'admin',
                data: {
                  id: authUser.id,
                  email: authUser.email,
                  nama_lengkap: 'Admin Pengawas'
                }
              };
              state.activeTab = 'monitor';
              await fetchAllDataFromSupabase();
              renderApp();
              return;
            }
          } catch (err) {
            console.warn('Pemeriksaan sesi admin dilewati:', err);
          }

          var savedPeserta = localStorage.getItem('lms_man4hst_peserta_session');
          if (savedPeserta) {
            try {
              var parsed = JSON.parse(savedPeserta);
              var checkRes = await supabase
                .from('peserta')
                .select('*')
                .eq('id', parsed.id)
                .eq('status_akun', 'aktif')
                .single();

              if (checkRes.data) {
                state.user = { role: 'peserta', data: checkRes.data };
                state.activeTab = 'beranda';
                await fetchAllDataFromSupabase();
                renderApp();
                return;
              } else {
                localStorage.removeItem('lms_man4hst_peserta_session');
              }
            } catch (e) {
              localStorage.removeItem('lms_man4hst_peserta_session');
            }
          }
        }

        renderLoginView();
      }

      async function handlePesertaLogin(e) {
        e.preventDefault();
        window.enterAppFullscreen();
        var formData = new FormData(e.target);
        var username = (formData.get('username') || '').toString().trim();
        var password = (formData.get('password') || '').toString().trim();

        if (!username || !password) {
          showToast('Harap isi NISN/Username dan Password.', 'warning');
          return;
        }

        if (IS_PREVIEW) {
          var found = state.pesertaList.find(function (p) {
            return p.username === username && p.password === password;
          });
          if (!found) {
            showToast('Username atau Password tidak cocok (Mode Preview).', 'error');
            return;
          }
          state.user = { role: 'peserta', data: found };
          state.activeTab = 'beranda';
          showToast('Selamat datang, ' + found.nama_lengkap, 'success');
          renderApp();
          return;
        }

        if (!supabase) {
          showToast('Konfigurasi SUPABASE_URL dan SUPABASE_ANON_KEY belum diisi.', 'error');
          return;
        }

        try {
          var res = await supabase
            .from('peserta')
            .select('*')
            .eq('username', username)
            .eq('password', password)
            .eq('status_akun', 'aktif')
            .single();

          if (res.error || !res.data) {
            showToast('NISN/Username atau Password salah, atau akun tidak aktif.', 'error');
            return;
          }

          localStorage.setItem('lms_man4hst_peserta_session', JSON.stringify({ id: res.data.id }));

          showToast('Berhasil masuk sebagai ' + res.data.nama_lengkap, 'success');
          renderApp();
        } catch (err) {
          console.error(err);
          showToast('Terjadi gangguan saat memverifikasi akun peserta.', 'error');
        }

      // ===================================================================
      // PROTEKSI ANTI CURANG (DETEKSI PINDAH TAB / KEHILANGAN FOKUS)
      // ===================================================================
      function activateAntiCheat() {
        state.antiCheatActive = true;
        document.body.classList.add('exam-lock-active');
        document.addEventListener('visibilitychange', handleVisibilityChange);
        window.addEventListener('blur', handleWindowBlur);
        document.addEventListener('contextmenu', blockContextMenu);
        document.addEventListener('keydown', blockShortcutKeys);
      }

      function deactivateAntiCheat() {
        state.antiCheatActive = false;
        document.body.classList.remove('exam-lock-active');
        document.removeEventListener('visibilitychange', handleVisibilityChange);
        window.removeEventListener('blur', handleWindowBlur);
        document.removeEventListener('contextmenu', blockContextMenu);
        document.removeEventListener('keydown', blockShortcutKeys);
        if (state.examTimerInterval) {
          clearInterval(state.examTimerInterval);
          state.examTimerInterval = null;
        }
      }

      function blockContextMenu(e) {
        if (state.antiCheatActive) e.preventDefault();
      }

      function blockShortcutKeys(e) {
        if (!state.antiCheatActive) return;
        if (
          e.key === 'F12' ||
          (e.ctrlKey && e.shiftKey && ['I', 'J', 'C'].indexOf(e.key.toUpperCase()) !== -1) ||
          (e.ctrlKey && e.key.toUpperCase() === 'U')
        ) {
          e.preventDefault();
          recordAntiCheatViolation('Mencoba membuka Inspect Element / Source Code');
        }
      }

      function handleVisibilityChange() {
        if (state.antiCheatActive && document.hidden) {
          recordAntiCheatViolation('Berpindah tab atau meminimalkan browser saat ujian');
        }
      }

      function handleWindowBlur() {
        if (state.antiCheatActive) {
          recordAntiCheatViolation('Fokus jendela berpindah keluar dari halaman ujian');
        }
      }

      async function recordAntiCheatViolation(reason) {
        if (!state.activeExamSession || !state.antiCheatActive) return;
        var now = Date.now();
        if (now - state.lastViolationTimestamp < 3000) return;
        state.lastViolationTimestamp = now;

        var monitoring = state.activeExamSession.monitoring;
        var sesi = state.activeExamSession.sesi;
        if (monitoring.status_ujian === 'terkunci' || monitoring.status_ujian === 'selesai') return;

        var newCount = (monitoring.jumlah_pelanggaran || 0) + 1;
        var maxAllowed = sesi.maks_pelanggaran || 3;
        var shouldLock = newCount >= maxAllowed;
        var nextStatus = shouldLock ? 'terkunci' : 'sedang_ujian';

        monitoring.jumlah_pelanggaran = newCount;
        monitoring.status_ujian = nextStatus;

        if (IS_PREVIEW) {
          for (var i = 0; i < state.monitoringList.length; i++) {
            if (state.monitoringList[i].id === monitoring.id) {
              state.monitoringList[i] = Object.assign({}, monitoring);
            }
          }
          state.logPelanggaranList.unshift({
            id: 'log-' + Date.now(),
            monitoring_id: monitoring.id,
            peserta_id: state.user.data.id,
            jenis_pelanggaran: reason,
            waktu_kejadian: new Date().toISOString()
          });
        } else if (supabase && sesi.id !== 'quick-portal-session') {
          try {
            await supabase
              .from('monitoring_ujian')
              .update({ jumlah_pelanggaran: newCount, status_ujian: nextStatus })
              .eq('id', monitoring.id);

            await supabase.from('log_pelanggaran').insert([
              {
                monitoring_id: monitoring.id,
                peserta_id: state.user.data.id,
                jenis_pelanggaran: reason
              }
            ]);
          } catch (err) {
            console.error('Gagal menyimpan log pelanggaran:', err);
          }
        }

        if (shouldLock) {
          deactivateAntiCheat();
          showToast('UJIAN TERKUNCI! Batas pelanggaran (' + maxAllowed + 'x) tercapai.', 'error');
        } else {
          showToast('PERINGATAN PELANGGARAN (' + newCount + '/' + maxAllowed + '): ' + reason, 'warning');
        }

        renderExamWorkspace();
      }

      function openPortalDirectly(url) {
        var targetUrl = url || DEFAULT_PORTAL_URL;
        var win = window.open(targetUrl, '_blank', 'noopener,noreferrer');
        if (!win) {
          var anchor = document.createElement('a');
          anchor.href = targetUrl;
          anchor.target = '_blank';
          anchor.rel = 'noopener noreferrer';
          document.body.appendChild(anchor);
          anchor.click();
          document.body.removeChild(anchor);
        }
      }

      window.checkUnlockStatus = async function () {
        if (!state.activeExamSession) return;
        var monitoring = state.activeExamSession.monitoring;

        if (IS_PREVIEW) {
          var latest = state.monitoringList.find(function (m) { return m.id === monitoring.id; });
          if (latest && latest.status_ujian !== 'terkunci') {
            state.activeExamSession.monitoring = latest;
            activateAntiCheat();
            renderExamWorkspace();
            showToast('Kunci ujian telah dibuka oleh Pengawas!', 'success');
          } else {
            showToast('Status ujian masih terkunci.', 'warning');
          }
          return;
        }

        if (supabase) {
          var monRes = await supabase.from('monitoring_ujian').select('*').eq('id', monitoring.id).single();
          if (monRes.data && monRes.data.status_ujian !== 'terkunci') {
            state.activeExamSession.monitoring = monRes.data;
            activateAntiCheat();
            renderExamWorkspace();
            showToast('Kunci ujian telah dibuka oleh Pengawas!', 'success');
          } else {
            showToast('Status ujian masih terkunci. Silakan hubungi Pengawas.', 'warning');
          }
        }
      };

      window.exitLockedScreen = function () {
        deactivateAntiCheat();
        state.activeExamSession = null;
        renderApp();

          '<div class="bg-white rounded-3xl max-w-md w-full p-6 shadow-2xl">' +
            '<h3 class="text-lg font-extrabold text-slate-800 mb-4">Tambah Materi / Pengumuman</h3>' +
            '<form id="form-materi" class="space-y-3.5">' +
              '<div><label class="block text-xs font-bold text-slate-700 mb-1">Judul</label><input type="text" name="judul" required placeholder="Judul pengumuman atau materi" class="w-full px-4 py-2.5 rounded-xl bg-slate-50 border border-slate-200 text-sm" /></div>' +
              '<div><label class="block text-xs font-bold text-slate-700 mb-1">Kategori</label><input type="text" name="kategori" required placeholder="Peraturan / Umum" class="w-full px-4 py-2.5 rounded-xl bg-slate-50 border border-slate-200 text-sm" /></div>' +
              '<div><label class="block text-xs font-bold text-slate-700 mb-1">Konten</label><textarea name="konten" rows="4" required placeholder="Tulis isi materi atau tata tertib..." class="w-full px-4 py-2.5 rounded-xl bg-slate-50 border border-slate-200 text-sm"></textarea></div>' +
              '<div class="flex gap-2.5 pt-3"><button type="button" onclick="closeModal()" class="flex-1 py-3 rounded-full bg-slate-100 text-slate-600 font-bold text-xs">Batal</button><button type="submit" class="flex-1 py-3 rounded-full bg-hero-gradient text-white font-bold text-xs shadow-lg">Publikasikan</button></div>' +
            '</form>' +
          '</div>'
        );

        document.getElementById('form-materi').addEventListener('submit', async function (ev) {
          ev.preventDefault();
          var fd = new FormData(ev.target);
          var payload = {
            judul: (fd.get('judul') || '').toString().trim(),
            kategori: (fd.get('kategori') || '').toString().trim(),
            konten: (fd.get('konten') || '').toString().trim()
          };

          if (IS_PREVIEW) {
            state.materiList.unshift(Object.assign({ id: 'm-' + Date.now() }, payload));
          } else if (supabase) {
            await supabase.from('materi_info').insert([payload]);
            await fetchAllDataFromSupabase();
          }

          closeModal();
          showToast('Materi/Info berhasil dipublikasikan.', 'success');
          renderApp();
        });
      };

      window.deleteMateri = async function (id) {
        if (IS_PREVIEW) {
          state.materiList = state.materiList.filter(function (m) { return m.id !== id; });
        } else if (supabase) {
          await supabase.from('materi_info').delete().eq('id', id);
          await fetchAllDataFromSupabase();
        }
        showToast('Materi dihapus.', 'info');
        renderApp();
      };

      initAuthSession();
    })();
  </script>
